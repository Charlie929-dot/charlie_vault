### for的取等条件
```c++
bool is_prime(int a)

{

    if(a<2) return false;

    else{

        for(int i=2;i*i<=a;i++){//=!!!!!!!!,或者i<a!!!!!!!!!

            if(a%i==0) return false;

        }

        return true;

    }

}
```

### 非void函数执行的每一个分支都要有return
```c++
int gcd(int a, int b){
	if(a==0) return b;
	else if(b==0) return a;
	else{
		if(a==b) return a;
		return gcd(b,a%b);
	}
}
```
