### for的取等条件
```c++
bool is_prime(int a)

{

    if(a<2) return false;

    else{

        for(int i=2;i*i<=a;i++){//=!!!!!!!!,或者i<a!!!!!!!!!

            if(a%i==0) return false;

        }

        return true;

    }

}
```
